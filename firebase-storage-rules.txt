// Firebase Storage Security Rules
// Copy these rules to your Firebase Console → Storage → Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Profile pictures - users can upload their own
    match /profile_pictures/{userId}_{timestamp}.jpg {
      allow read: if true; // Anyone can view profile pictures
      allow write: if request.auth != null 
                   && request.auth.uid == userId
                   && request.resource.size < 5 * 1024 * 1024 // 5MB limit
                   && request.resource.contentType.matches('image/.*');
    }
    
    // Catch-all: deny all other access
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
